############################################
# JAGS model
# Poisson GLARMA(1,1) with Spike-slab priors
############################################

model {

  # likelihood
  mu[1] <- y_star[1]
  z[1] <- 0

  for(i in 2:N){

    z[i] <- phi * (log(y_star[i-1]) - inprod(X_alpha[i-1, ], alpha) - inprod(X_beta[i-1, ], beta)) +
        theta * log(y_star[i-1]/mu[i-1])

    mu[i] <- exp(inprod(X_alpha[i, ], alpha) + inprod(X_beta[i, ], beta) + z[i])

  }

  for(i in 2:N){
    y[i] ~ dpois(mu[i])
  }


  # priors
  alpha ~ dnorm(0, 0.2)
  phi ~ dbeta(1, 1)
  theta ~ dbeta(1, 1)

  for(p in 1:P){
    gamma[p] <- (1 - ind[p]) * nu0 + ind[p] * nu1
    beta[p] <- beta_std[p] * gamma[p]
    ind[p] ~ dbern(pi0)
    beta_std[p] ~ dnorm(0, 1)
  }


}
